# ------------------------------------------------------------------ doodles
def arrow(x1, y1, x2, y2, color=INK, w=4, head=13, speed=2.0):
    a = math.atan2(y2 - y1, x2 - x1)
    return [line(x1, y1, x2, y2, w=w, color=color, speed=speed),
            poly([(x2 - head * math.cos(a - .45), y2 - head * math.sin(a - .45)), (x2, y2),
                  (x2 - head * math.cos(a + .45), y2 - head * math.sin(a + .45))],
                 w=w, color=color, speed=3.0)]


def box(x, y, w, h, label, fnt=SMALL, color=INK, tc=INK, speed=2.2):
    tw = fnt.getbbox(label)[2]
    return [rect(x, y, w, h, w=4, color=color, speed=speed),
            text(x + (w - tw) / 2, y + h / 2 - fnt.size * 0.62, label, fnt, tc, speed=1.5)]


def stickman(x, y, s=1.0, color=INK):
    return [circle(x, y, 13 * s, n=16, w=4, color=color, speed=2.5),
            line(x, y + 13 * s, x, y + 48 * s, w=4, color=color, speed=3),
            poly([(x - 20 * s, y + 42 * s), (x, y + 24 * s), (x + 20 * s, y + 40 * s)], w=4, color=color, speed=3),
            poly([(x - 16 * s, y + 80 * s), (x, y + 48 * s), (x + 16 * s, y + 80 * s)], w=4, color=color, speed=3)]


def crate(x, y, s=1.0, color=INK):

def road_persp(x, y, w, h, color=INK):
    """Road running to a vanishing point."""
    out = [poly([(x, y + h), (x + w * 0.36, y), (x + w * 0.64, y), (x + w, y + h)], w=5, color=color, speed=1.6)]
    for i, f in enumerate((0.18, 0.42, 0.68, 0.9)):
        yy = y + h * f
        hw = 4 + 14 * f
        out.append(line(x + w / 2 - hw * 0.2, yy, x + w / 2 + hw * 0.2, yy + h * 0.10,
                        w=int(3 + 5 * f), color=ORANGE, speed=4))
    return out


def layers(x, y, w, color=INK):
    """Pavement cross-section: surface, base, sub-base, subgrade."""
    hs = [20, 26, 30, 38]
    names = ["surface", "base", "sub-base", "subgrade"]
    out, yy = [], y
    for h, nm in zip(hs, names):
        out.append(rect(x, yy, w, h, w=3, color=color, speed=3.0))
        out.append(text(x + w + 10, yy + h / 2 - 12, nm, TINY, GREY, speed=2.6))
        yy += h
    return out

def culvert(x, y, color=INK):
    return [line(x, y, x + 96, y, w=4, color=color, speed=3),

                 w=4, color=BLUE, speed=3)]


def curve_geom(x, y, color=INK):
    pts = [(x + t * 130, y + 46 * math.sin(t * 2.6) * (1 - t * 0.35)) for t in

def hl(x, y, w, color=(252, 214, 88)):
    """Marker highlight bar (drawn before the text that sits on it)."""
    return poly([(x, y), (x + w, y)], w=26, color=color, speed=1.8)

import math

FD = os.path.join(os.path.dirname(os.path.abspath(__file__)), "fonts")


def font(name, size, bold=False):
    f = ImageFont.truetype(os.path.join(FD, name + ".ttf"), size)
    if bold:
        try:
            f.set_variation_by_name("Bold")
        except Exception:
            pass
    return f


HAND_SRC = "/home/claude/hand.png"
HAND_TIP = (237, 220)          # marker nib in the source image
HAND_W = 430                   # on-canvas width


def load_hand():
    im = Image.open(HAND_SRC).convert("RGBA")
    k = HAND_W / im.width
    im = im.resize((int(im.width * k), int(im.height * k)), Image.LANCZOS)
    return im, (HAND_TIP[0] * k, HAND_TIP[1] * k)


# ---------------------------------------------------------------- stroke types

def poly(points, w=4, color=INK, speed=1.0, close=False):
    pts = list(points) + ([points[0]] if close else [])
    return dict(kind="poly", pts=pts, w=w, color=color, speed=speed)


def line(x1, y1, x2, y2, **kw):
    return poly([(x1, y1), (x2, y2)], **kw)


def rect(x, y, bw, bh, **kw):
    return poly([(x, y), (x + bw, y), (x + bw, y + bh), (x, y + bh)], close=True, **kw)


def circle(cx, cy, r, n=30, a0=0, a1=360, **kw):
    pts = [(cx + r * math.cos(math.radians(a0 + (a1 - a0) * i / n)),
            cy + r * math.sin(math.radians(a0 + (a1 - a0) * i / n))) for i in range(n + 1)]
    return poly(pts, **kw)


def ellipse(cx, cy, rx, ry, n=36, **kw):
    pts = [(cx + rx * math.cos(math.radians(360 * i / n)),
            cy + ry * math.sin(math.radians(360 * i / n))) for i in range(n + 1)]
    return poly(pts, **kw)


def check(x, y, s=1.0, color=(198, 58, 48), w=5):
    return poly([(x, y + 12 * s), (x + 10 * s, y + 24 * s), (x + 30 * s, y)],
                w=w, color=color, speed=2.4)


def text(x, y, s, fnt, color=INK, speed=1.0):
    return dict(kind="text", x=x, y=y, s=s, font=fnt, color=color, speed=speed)


def texts(x, y, lines, fnt, color=INK, lh=None, speed=1.0):
    lh = lh or int(fnt.size * 1.45)
    return [text(x, y + i * lh, s, fnt, color, speed) for i, s in enumerate(lines)]


def bullets(x, y, lines, fnt, color=INK, lh=None, dot=RED, speed=1.0):
    lh = lh or int(fnt.size * 1.55)
    out = []
    for i, s in enumerate(lines):
        yy = y + i * lh
        out.append(circle(x - 14, yy + fnt.size * 0.62, 4.0, n=10, w=4, color=dot, speed=2.6))
        out.append(text(x, yy, s, fnt, color, speed))
    return out


def measure(s, fnt):
    b = fnt.getbbox(s)
    return b[2] - b[0], b[3] - b[1]


# ------------------------------------------------------------------- rendering

def _seglens(pts):
    L = [0.0]
    for i in range(1, len(pts)):
        L.append(L[-1] + math.dist(pts[i - 1], pts[i]))
    return L


def _draw_poly(layer, st, t):
    pts, L = st["pts"], _seglens(st["pts"])
    total = L[-1]
    if total == 0:
        return pts[0]
    target = total * t
    d = ImageDraw.Draw(layer)
    out = [pts[0]]
    for i in range(1, len(pts)):
        if L[i] <= target:
            out.append(pts[i])
        else:
            f = (target - L[i - 1]) / max(1e-6, L[i] - L[i - 1])
            out.append((pts[i - 1][0] + (pts[i][0] - pts[i - 1][0]) * f,
                        pts[i - 1][1] + (pts[i][1] - pts[i - 1][1]) * f))
            break
    if len(out) > 1:
        d.line(out, fill=st["color"] + (255,), width=st["w"], joint="curve")
        r = st["w"] / 2.0
        for p in (out[0], out[-1]):
            d.ellipse([p[0] - r, p[1] - r, p[0] + r, p[1] + r], fill=st["color"] + (255,))
    return out[-1]


def _text_img(st):
    b = st["font"].getbbox(st["s"])
    tmp = Image.new("RGBA", (max(2, b[2] + 4), max(2, b[3] + 6)), (0, 0, 0, 0))
    ImageDraw.Draw(tmp).text((0, 0), st["s"], font=st["font"], fill=st["color"] + (255,))
    return tmp, b


def _draw_text(layer, st, t):
    tmp, b = _text_img(st)
    reveal = max(1, int(tmp.width * t))
    layer.paste(tmp.crop((0, 0, reveal, tmp.height)), (int(st["x"]), int(st["y"])))
    return (st["x"] + reveal, st["y"] + b[3] * 0.82)


def stroke_len(st):
    if st["kind"] == "poly":
        return _seglens(st["pts"])[-1]
    return _text_img(st)[0].width


# ---------------------------------------------------------------------- board

class Whiteboard:
    def __init__(self, outdir):
        self.dir = outdir
        os.makedirs(outdir, exist_ok=True)
        for f in os.listdir(outdir):
            os.remove(os.path.join(outdir, f))
        self.blank = self._bg()
        self.board = self.blank.copy()
        self.hand, self.tip = load_hand()
        self.n = 0

    def _bg(self):
        img = Image.new("RGB", (W, H), (255, 255, 255))
        px = img.load()
        cx, cy, mx = W * 0.5, H * 0.45, math.hypot(W * 0.5, H * 0.5)
        for y in range(H):
            for x in range(0, W, 2):
                v = 255 - int(26 * (math.hypot(x - cx, y - cy) / mx) ** 2.1)
                px[x, y] = (v, v, v - 1)
                px[min(W - 1, x + 1), y] = (v, v, v - 1)
        d = ImageDraw.Draw(img)
        d.rectangle([0, 0, W - 1, H - 1], outline=(214, 214, 214), width=3)
        return img

    def emit(self, layer=None, tip=None):
        img = self.board if layer is None else self.board.copy()
        if layer is not None:
            img.paste(layer, (0, 0), layer)
        if tip is not None:
            img = img.convert("RGBA")
            img.alpha_composite(self.hand, (int(tip[0] - self.tip[0]), int(tip[1] - self.tip[1])))
            img = img.convert("RGB")
        img.save(f"{self.dir}/f{self.n:05d}.png")
        self.n += 1

    def bake(self, st):
        layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
        (_draw_poly if st["kind"] == "poly" else _draw_text)(layer, st, 1.0)
        self.board.paste(layer, (0, 0), layer)

    def play(self, strokes, px_per_frame=100.0, text_px_per_frame=58.0):
        for st in strokes:
            if st.get("kind") == "pause":
                self.hold(st["t"])
                continue
            base = text_px_per_frame if st["kind"] == "text" else px_per_frame
            n = max(2, int(round(stroke_len(st) / (base * st.get("speed", 1.0)))))
            fn = _draw_poly if st["kind"] == "poly" else _draw_text
            for k in range(1, n + 1):
                layer = Image.new("RGBA", (W, H), (0, 0, 0, 0))
                tip = fn(layer, st, k / n)
                self.emit(layer, tip)
            self.bake(st)

    def hold(self, seconds):
        for _ in range(max(1, int(seconds * FPS))):
            self.emit()

    def wipe(self, seconds=0.45):
        n = max(3, int(seconds * FPS))
        start = self.board.copy()
        for k in range(1, n + 1):
            img = start.copy()
            x = int(W * k / n)
            img.paste(self.blank.crop((0, 0, x, H)), (0, 0))
            img.save(f"{self.dir}/f{self.n:05d}.png")
            self.n += 1
        self.board = self.blank.copy()


def pause(t):
    return dict(kind="pause", t=t)
